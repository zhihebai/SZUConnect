# SZUConnect for Windows (native C++20, Win32 + Direct2D, pinned static libcurl/Schannel).
#   Windows:  cmake --preset windows-x64 && cmake --build --preset windows-release && ctest --preset windows-release
#   Any OS:   cmake --preset core-only && cmake --build --preset core && ctest --preset core   (portable core tests only)
cmake_minimum_required(VERSION 3.21)
project(SZUConnect VERSION 1.0.0.33 DESCRIPTION "SZUConnect campus network helper (unofficial)" LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# Static CRT: the single SZUConnect.exe runs without a VC++ redistributable.
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc /Zc:__cplusplus /sdl /MP)
  include(CheckIPOSupported)
  check_ipo_supported(RESULT SZU_IPO_SUPPORTED OUTPUT SZU_IPO_ERROR LANGUAGES C CXX)
  if(SZU_IPO_SUPPORTED)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
  else()
    message(STATUS "Release interprocedural optimization unavailable: ${SZU_IPO_ERROR}")
  endif()
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()

# ---- portable core (no Windows headers; also built and tested on Linux/macOS) ----
file(GLOB SZU_CORE_SOURCES CONFIGURE_DEPENDS core/src/*.cpp core/generated/*.cpp)
add_library(szu_core STATIC ${SZU_CORE_SOURCES})
target_include_directories(szu_core PUBLIC core/include core/generated)

include(CTest)
if(BUILD_TESTING)
  add_executable(szu_core_tests tests/core_tests.cpp)
  target_link_libraries(szu_core_tests PRIVATE szu_core)
  # The argument is the repository root that contains shared/ (fixtures, copy, tokens).
  add_test(NAME szu_core_tests COMMAND szu_core_tests "${CMAKE_CURRENT_SOURCE_DIR}/../..")
  add_executable(szu_info_popover_placement_tests tests/info_popover_placement_tests.cpp app/info_popover_placement.cpp)
  target_include_directories(szu_info_popover_placement_tests PRIVATE app)
  add_test(NAME szu_info_popover_placement_tests COMMAND szu_info_popover_placement_tests)
  set_tests_properties(szu_info_popover_placement_tests PROPERTIES TIMEOUT 10)
endif()

# ---- Windows platform layer + app ----
if(WIN32)
  enable_language(RC)
  # Fixed source/hash, static CRT and OS certificate validation. No external runtime DLL.
  include(FetchContent)
  set(BUILD_CURL_EXE OFF CACHE BOOL "" FORCE)
  set(BUILD_LIBCURL_DOCS OFF CACHE BOOL "" FORCE)
  set(BUILD_MISC_DOCS OFF CACHE BOOL "" FORCE)
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  set(BUILD_STATIC_LIBS ON CACHE BOOL "" FORCE)
  set(CURL_STATIC_CRT ON CACHE BOOL "" FORCE)
  set(CURL_USE_SCHANNEL ON CACHE BOOL "" FORCE)
  set(CURL_USE_OPENSSL OFF CACHE BOOL "" FORCE)
  set(CURL_ZLIB OFF CACHE BOOL "" FORCE)
  set(CURL_BROTLI OFF CACHE BOOL "" FORCE)
  set(CURL_ZSTD OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBPSL OFF CACHE BOOL "" FORCE)
  set(CURL_USE_LIBSSH2 OFF CACHE BOOL "" FORCE)
  set(USE_NGHTTP2 OFF CACHE BOOL "" FORCE)
  set(CURL_DISABLE_INSTALL ON CACHE BOOL "" FORCE)
  foreach(protocol DICT FILE FTP GOPHER IMAP LDAP LDAPS MQTT POP3 RTSP SMB SMTP TELNET TFTP)
    set(CURL_DISABLE_${protocol} ON CACHE BOOL "" FORCE)
  endforeach()
  FetchContent_Declare(szu_curl
    URL https://curl.se/download/curl-8.22.0.tar.xz
    URL_HASH SHA256=f7ef3ae8a22e521f289803fe93543eb64c329b58aa73a9e224dfd915a2a5f4f7
  )
  set(_szu_build_testing "${BUILD_TESTING}")
  set(BUILD_TESTING OFF)
  FetchContent_MakeAvailable(szu_curl)
  set(BUILD_TESTING "${_szu_build_testing}")

  file(GLOB SZU_PLATFORM_SOURCES CONFIGURE_DEPENDS platform/src/*.cpp)
  add_library(szu_platform STATIC ${SZU_PLATFORM_SOURCES})
  target_include_directories(szu_platform PUBLIC platform/include)
  target_compile_definitions(szu_platform PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(szu_platform PUBLIC szu_core CURL::libcurl winhttp iphlpapi ws2_32 advapi32 shell32 ole32 user32 windowsapp wlanapi)

  if(BUILD_TESTING)
    add_executable(szu_installation_tests tests/installation_tests.cpp app/installation.cpp)
    target_include_directories(szu_installation_tests PRIVATE app)
    target_compile_definitions(szu_installation_tests PRIVATE SZU_INSTALLATION_TESTING UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(szu_installation_tests PRIVATE bcrypt version advapi32 shell32 ole32 user32)
    add_test(NAME szu_installation_tests COMMAND szu_installation_tests)
    set_tests_properties(szu_installation_tests PROPERTIES TIMEOUT 30)
    add_executable(szu_info_popover_native_tests tests/info_popover_native_tests.cpp
      app/info_popover.cpp app/info_popover_placement.cpp app/accessibility.cpp app/canvas.cpp
      app/scene.cpp app/theme.cpp app/brand_icon.cpp resources/app.manifest)
    target_include_directories(szu_info_popover_native_tests PRIVATE app platform/include)
    target_compile_definitions(szu_info_popover_native_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(szu_info_popover_native_tests PRIVATE szu_core d2d1 dwrite windowscodecs uiautomationcore oleaut32 oleacc ole32 comctl32 dwmapi gdi32 user32)
    if(MSVC)
      target_link_options(szu_info_popover_native_tests PRIVATE /MANIFESTUAC:NO)
    endif()
    add_test(NAME szu_info_popover_native_tests COMMAND szu_info_popover_native_tests)
    set_tests_properties(szu_info_popover_native_tests PROPERTIES TIMEOUT 30)
    add_executable(szu_records_view_native_tests tests/records_view_native_tests.cpp
      app/accessibility.cpp app/canvas.cpp app/scene.cpp app/theme.cpp app/brand_icon.cpp resources/app.manifest)
    target_include_directories(szu_records_view_native_tests PRIVATE app platform/include)
    target_compile_definitions(szu_records_view_native_tests PRIVATE SZU_VIEW_TESTING UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00 WINVER=0x0A00)
    target_link_libraries(szu_records_view_native_tests PRIVATE szu_core d2d1 dwrite windowscodecs uiautomationcore oleaut32 oleacc ole32 comctl32 dwmapi gdi32 user32)
    if(MSVC)
      target_link_options(szu_records_view_native_tests PRIVATE /MANIFESTUAC:NO)
    endif()
    add_test(NAME szu_records_view_native_tests COMMAND szu_records_view_native_tests)
    set_tests_properties(szu_records_view_native_tests PROPERTIES TIMEOUT 20)
    add_executable(szu_runtime_tests tests/runtime_tests.cpp app/runtime.cpp app/controller.cpp)
    target_include_directories(szu_runtime_tests PRIVATE app)
    target_link_libraries(szu_runtime_tests PRIVATE szu_platform)
    add_test(NAME szu_runtime_tests COMMAND szu_runtime_tests)
    set_tests_properties(szu_runtime_tests PROPERTIES TIMEOUT 30)
    add_executable(szu_physical_service_tests tests/physical_service_tests.cpp)
    target_link_libraries(szu_physical_service_tests PRIVATE szu_platform)
    add_test(NAME szu_physical_service_tests COMMAND szu_physical_service_tests)
    set_tests_properties(szu_physical_service_tests PROPERTIES TIMEOUT 30)
    add_executable(szu_physical_http_tests tests/physical_http_native_tests.cpp platform/src/physical_http.cpp)
    target_compile_definitions(szu_physical_http_tests PRIVATE SZU_PHYSICAL_HTTP_TESTING)
    target_link_libraries(szu_physical_http_tests PRIVATE szu_platform)
    add_test(NAME szu_physical_http_tests COMMAND szu_physical_http_tests)
    set_tests_properties(szu_physical_http_tests PROPERTIES TIMEOUT 30)
    add_executable(szu_physical_http_cancellation_tests tests/physical_http_cancellation_tests.cpp platform/src/physical_http.cpp)
    target_compile_definitions(szu_physical_http_cancellation_tests PRIVATE SZU_PHYSICAL_HTTP_TESTING)
    target_link_libraries(szu_physical_http_cancellation_tests PRIVATE szu_platform)
    add_test(NAME szu_physical_http_cancellation_tests COMMAND szu_physical_http_cancellation_tests)
    set_tests_properties(szu_physical_http_cancellation_tests PROPERTIES TIMEOUT 30)
  endif()

  file(GLOB SZU_APP_SOURCES CONFIGURE_DEPENDS app/*.cpp app/*.hpp)
  set(SZU_INSTALL_README "${CMAKE_CURRENT_SOURCE_DIR}/../../docs/PACKAGE-WINDOWS.md")
  set(SZU_INSTALL_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE")
  set(SZU_INSTALL_NOTICES "${CMAKE_CURRENT_SOURCE_DIR}/../../docs/DISTRIBUTION-NOTICES.txt")
  set(SZU_INSTALL_CURL_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/../../docs/licenses/curl-COPYING.txt")
  configure_file(resources/installation-documents.rc.in "${CMAKE_CURRENT_BINARY_DIR}/installation-documents.rc" @ONLY)
  set_source_files_properties("${CMAKE_CURRENT_BINARY_DIR}/installation-documents.rc" PROPERTIES
    OBJECT_DEPENDS "${SZU_INSTALL_README};${SZU_INSTALL_LICENSE};${SZU_INSTALL_NOTICES};${SZU_INSTALL_CURL_LICENSE}")
  add_executable(SZUConnect WIN32 ${SZU_APP_SOURCES} resources/SZUConnect.rc resources/app.manifest
    "${CMAKE_CURRENT_BINARY_DIR}/installation-documents.rc")
  target_include_directories(SZUConnect PRIVATE app resources)
  target_link_libraries(SZUConnect PRIVATE szu_platform d2d1 dwrite windowscodecs uiautomationcore oleaut32 oleacc comctl32 dwmapi gdi32 version bcrypt)
  if(MSVC)
    # app.manifest carries trustInfo (asInvoker), PerMonitorV2 DPI and Common Controls v6.
    target_link_options(SZUConnect PRIVATE /MANIFESTUAC:NO)
  endif()
  set_target_properties(SZUConnect PROPERTIES OUTPUT_NAME "SZUConnect")
endif()

if(BUILD_TESTING AND NOT WIN32)
  add_executable(szu_physical_service_tests tests/physical_service_tests.cpp platform/src/service.cpp)
  target_include_directories(szu_physical_service_tests PRIVATE platform/include)
  target_link_libraries(szu_physical_service_tests PRIVATE szu_core)
  add_test(NAME szu_physical_service_tests COMMAND szu_physical_service_tests)
  set_tests_properties(szu_physical_service_tests PROPERTIES TIMEOUT 30)
endif()
